```csharp
public async Task<StreamsEventResponse> Handler(DynamoDBEvent evt)
{
    var failures = new List<StreamsEventResponse.BatchItemFailure>();
    foreach (var r in evt.Records)
        try { await Notify(r); }
        catch { failures.Add(new() { ItemIdentifier = r.Dynamodb.SequenceNumber }); }
    return new StreamsEventResponse { BatchItemFailures = failures };
}
```
